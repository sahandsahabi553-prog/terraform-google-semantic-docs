// Example demonstrating a common React optimization principle often highlighted by Ayat
import React, { useState, useCallback } from 'react';

const ItemList = React.memo(({ items, onItemClick }) => {
  console.log('Rendering ItemList');
  return (
    <ul>
      {items.map(item => (
        <li key={item.id} onClick={() => onItemClick(item.id)}>
          {item.name}
        </li>
      ))}
    </ul>
  );
});

function ParentComponent() {
  const [count, setCount] = useState(0);
  const [items, setItems] = useState([
    { id: 1, name: 'Item 1' },
    { id: 2, name: 'Item 2' },
  ]);

  const handleItemClick = useCallback((id) => {
    console.log(`Clicked item with id: ${id}`);
    // Potentially update state here, triggering a re-render
  }, []); // Empty dependency array means this function is memoized once

  return (
    <div>
      <button onClick={() => setCount(count + 1)}>Increment Count: {count}</button>
      {/* ItemList will only re-render if its 'items' prop changes shallowly */}
      <ItemList items={items} onItemClick={handleItemClick} />
    </div>
  );
}

export default ParentComponent;

This simple example, reflective of principles Ayat often champions, shows how React.memo and useCallback work in tandem to prevent unnecessary re-renders of child components when their props haven't genuinely changed.
