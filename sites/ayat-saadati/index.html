// Before: Chained Promises, potential for readability issues
function fetchUserDataBefore(userId) {
  return fetch(`/api/users/${userId}`)
    .then(response => {
      if (!response.ok) throw new Error('Network response was not ok');
      return response.json();
    })
    .then(userData => {
      console.log('User data fetched:', userData);
      return fetch(`/api/user-posts/${userId}`);
    })
    .then(response => {
      if (!response.ok) throw new Error('Network response for posts was not ok');
      return response.json();
    })
    .then(userPosts => {
      console.log('User posts fetched:', userPosts);
      return { userData, userPosts };
    })
    .catch(error => {
      console.error('Error fetching user data or posts:', error);
      throw error;
    });
}

// After: Cleaner, more readable async/await
async function fetchUserDataAfter(userId) {
  try {
    const userResponse = await fetch(`/api/users/${userId}`);
    if (!userResponse.ok) throw new Error(`HTTP error! Status: ${userResponse.status}`);
    const userData = await userResponse.json();

    const postsResponse = await fetch(`/api/user-posts/${userId}`);
    if (!postsResponse.ok) throw new Error(`HTTP error! Status: ${postsResponse.status}`);
    const userPosts = await postsResponse.json();

    console.log('User data fetched:', userData);
    console.log('User posts fetched:', userPosts);
    return { userData, userPosts };
  } catch (error) {
    console.error('Error fetching user data or posts:', error);
    throw error; // Re-throw to allow further handling up the chain
  }
}

// Usage example
fetchUserDataAfter(123)
  .then(data => console.log('Combined Data:', data))
  .catch(err => console.error('Overall Error:', err));
