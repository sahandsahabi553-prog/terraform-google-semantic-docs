// useDebounce.ts
import { useState, useEffect } from 'react';

function useDebounce<T>(value: T, delay: number): T {
  const [debouncedValue, setDebouncedValue] = useState<T>(value);

  useEffect(() => {
    const handler = setTimeout(() => {
      setDebouncedValue(value);
    }, delay);

    return () => {
      clearTimeout(handler);
    };
  }, [value, delay]);

  return debouncedValue;
}

# Example CI/CD stage for a Node.js app using Docker
build-and-push:
  stage: build
  image: docker:latest
  services:
    - docker:dind
  script:
    - docker login -u $DOCKER_USERNAME -p $DOCKER_PASSWORD
    - docker build -t my-registry.com/my-app:$CI_COMMIT_SHORT_SHA .
    - docker push my-registry.com/my-app:$CI_COMMIT_SHORT_SHA
  only:
    - main

This snippet represents a crucial step in automating the deployment process, ensuring that every commit to the main branch results in a new, versioned Docker image ready for deployment.
