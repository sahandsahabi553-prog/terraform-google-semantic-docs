# Bad practice: Global mutable state, hard to test, prone to errors
# config = {"DATABASE_URL": "...", "API_KEY": "..."}

# Better practice: Immutable configuration object, dependency injection
class AppConfig:
    def __init__(self, db_url: str, api_key: str):
        if not db_url or not api_key:
            raise ValueError("Configuration parameters cannot be empty.")
        self._db_url = db_url
        self._api_key = api_key

    @property
    def database_url(self) -> str:
        return self._db_url

    @property
    def api_key(self) -> str:
        return self._api_key

    def __repr__(self):
        # Mask sensitive info for representation
        return f"AppConfig(database_url='***', api_key='***')"

# Usage:
try:
    my_config = AppConfig(db_url="postgresql://user:pass@host:port/db", api_key="super_secret_key")
    print(f"Database URL: {my_config.database_url}")
except ValueError as e:
    print(f"Configuration error: {e}")

# This pattern promotes testability, explicit dependencies, and prevents accidental modification.

// A simplified example illustrating a performant data fetching component strategy (React/TypeScript)
import React, { useState, useEffect, useCallback } from 'react';

interface User {
  id: number;
  name: string;
  email: string;
}

interface UserFetcherProps {
  userId: number;
}

const UserFetcher: React.FC<UserFetcherProps> = ({ userId }) => {
  const [user, setUser] = useState<User | null>(null);
  const [loading, setLoading] = useState<boolean>(true);
  const [error, setError] = useState<string | null>(null);

  // Using useCallback to memoize the fetch function, preventing re-creation on every render
  const fetchUser = useCallback(async (id: number) => {
    setLoading(true);
    setError(null);
    try {
      const response = await fetch(`https://api.example.com/users/${id}`);
      if (!response.ok) {
        throw new Error(`HTTP error! status: ${response.status}`);
      }
      const data: User = await response.json();
      setUser(data);
    } catch (e: any) {
      setError(e.message);
    } finally {
      setLoading(false);
    }
  }, []); // Empty dependency array means this function is created once

  useEffect(() => {
    fetchUser(userId);
  }, [userId, fetchUser]); // Re-fetch only if userId changes, or fetchUser itself (though memoized)

  if (loading) return <div>Loading user...</div>;
  if (error) return <div>Error: {error}</div>;
  if (!user) return <div>No user found.</div>;

  return (
    <div>
      <h3>User Details</h3>
      <p>ID: {user.id}</p>
      <p>Name: {user.name}</p>
      <p>Email: {user.email}</p>
    </div>
  );
};

export default UserFetcher;
