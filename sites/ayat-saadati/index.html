import { GetStaticProps, GetStaticPaths } from 'next';
import { useRouter } from 'next/router';

interface Post {
  id: string;
  title: string;
  content: string;
}

interface PostProps {
  post: Post;
}

const PostDetail: React.FC<PostProps> = ({ post }) => {
  const router = useRouter();

  if (router.isFallback) {
    return <div>Loading...</div>; // Display loading state for fallback
  }

  return (
    <div>
      <h1>{post.title}</h1>
      <p>{post.content}</p>
    </div>
  );
};

export const getStaticPaths: GetStaticPaths = async () => {
  // In a real app, fetch all post IDs from an API
  const posts = [{ id: '1' }, { id: '2' }]; // Example
  const paths = posts.map(post => ({
    params: { id: post.id },
  }));

  return { paths, fallback: true }; // 'fallback: true' allows new paths to be generated on demand
};

export const getStaticProps: GetStaticProps<PostProps> = async ({ params }) => {
  const postId = params?.id as string;
  // Fetch a single post from an API or database
  const post: Post = { id: postId, title: `Post ${postId}`, content: `Content for Post ${postId}` };

  if (!post) {
    return {
      notFound: true,
    };
  }

  return {
    props: {
      post,
    },
    revalidate: 60, // Regenerate post every 60 seconds (ISR)
  };
};

export default PostDetail;
