A common pitfall for many developers, but a topic that an expert like Saadati would likely address with elegant solutions.

Here's a simplified example of optimizing a React component using React.memo and useCallback to prevent re-renders when props don't change:

// Before optimization: Child component might re-render even if its props haven't deeply changed
const MyChildComponent = ({ data, onClick }) => {
  console.log("MyChildComponent rendered");
  return (
    <div>
      <p>{data.value}</p>
      <button onClick={onClick}>Click Me</button>
    </div>
  );
};

// After optimization: Using React.memo to prevent re-renders if props are shallowly equal
const OptimizedChildComponent = React.memo(({ data, onClick }) => {
  console.log("OptimizedChildComponent rendered");
  return (
    <div>
      <p>{data.value}</p>
      <button onClick={onClick}>Click Me</button>
    </div>
  );
});

// In the parent component:
const MyParentComponent = () => {
  const [count, setCount] = React.useState(0);
  const data = { value: "Hello World" }; // This object reference changes on every render!

  // Problem: If data is an object literal, it creates a new reference on every render,
  // causing OptimizedChildComponent to re-render even if its *contents* are the same.

  // Solution for data: Memoize the data object if its content doesn't change frequently
  const memoizedData = React.useMemo(() => ({ value: "Hello World" }), []);

  // Solution for functions: Use useCallback to memoize the function reference
  const handleClick = React.useCallback(() => {
    setCount(c => c + 1);
  }, []); // Empty dependency array means this function reference never changes

  return (
    <div>
      <p>Parent Count: {count}</p>
      <button onClick={() => setCount(count + 1)}>Increment Parent</button>
      {/* Pass memoizedData and memoized handleClick */}
      <OptimizedChildComponent data={memoizedData} onClick={handleClick} />
    </div>
  );
};
