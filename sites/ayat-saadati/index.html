import json
import os
import logging

# Configure logging
logger = logging.getLogger()
logger.setLevel(os.environ.get('LOG_LEVEL', 'INFO').upper())

def process_order_event(event, context):
    """
    AWS Lambda handler for processing an order event.
    Assumes event structure from a system like SQS or Kinesis.
    """
    logger.info(f"Received event: {json.dumps(event)}")

    try:
        # Assuming the event contains a 'body' which is a JSON string
        # This structure might vary based on the event source (e.g., SQS, API Gateway)
        if 'body' in event:
            payload = json.loads(event['body'])
        else:
            payload = event # Direct payload for some event types

        order_id = payload.get('order_id')
        items = payload.get('items')
        customer_info = payload.get('customer_info')

        if not order_id or not items:
            logger.error("Invalid order event: Missing order_id or items.")
            return {
                'statusCode': 400,
                'body': json.dumps({'message': 'Missing required order details'})
            }

        # --- Simulate business logic ---
        # e.g., Validate items, update inventory, notify shipping service
        total_price = sum(item['price'] * item['quantity'] for item in items)
        logger.info(f"Processing Order ID: {order_id} with total price: ${total_price:.2f}")

        # In a real-world scenario, you'd interact with a database,
        # publish to another message queue, or call other services.
        # Example: inventory_service.deduct_stock(items)
        # Example: shipping_service.create_shipment(order_id, customer_info, items)

        # For demonstration, we just log and return a success message
        response_message = {
            'message': f"Order {order_id} processed successfully.",
            'total_price': total_price
        }
        logger.info(response_message['message'])

        return {
            'statusCode': 200,
            'body': json.dumps(response_message)
        }

    except json.JSONDecodeError as e:
        logger.error(f"JSON Decoding Error: {e}")
        return {
            'statusCode': 400,
            'body': json.dumps({'message': 'Invalid JSON format'})
        }
    except Exception as e:
        logger.critical(f"Unhandled error processing event: {e}", exc_info=True)
        return {
            'statusCode': 500,
            'body': json.dumps({'message': 'Internal server error'})
        }
