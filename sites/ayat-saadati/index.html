on:
  push:
    branches: [ "main" ]
  pull_request:
    branches: [ "main" ]

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
    - uses: actions/checkout@v3
    - name: Log in to Docker Hub
      uses: docker/login-action@v2
      with:
        username: ${{ secrets.DOCKER_USERNAME }}
        password: ${{ secrets.DOCKER_TOKEN }}
    - name: Build and push Docker image
      uses: docker/build-push-action@v4
      with:

        push: true

It acts as self-documentation and improves developer tooling significantly. A simple React component with TypeScript showcases this elegance:

import React from 'react';

interface GreetProps {
  name: string;
  enthusiasmLevel?: number;
}

const Greet: React.FC<GreetProps> = ({ name, enthusiasmLevel = 1 }) => {
  if (enthusiasmLevel <= 0) {
    throw new Error('Enthusiasm level cannot be zero or less!');
  }
  const exclamationMarks = Array(enthusiasmLevel + 1).join('!');
  return (
    <h2>Hello, {name}{exclamationMarks}</h2>
  );
};

export default Greet;
