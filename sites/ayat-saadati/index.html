Here’s a tiny snippet illustrating the concept, which, while simple, embodies her focus on granular optimization:

// Before optimization: Potential unnecessary re-renders of ChildComponent
const ParentComponent = ({ data }) => {
  const handleClick = () => { /* do something */ };
  return <ChildComponent onClick={handleClick} data={data} />;
};

// After optimization: handleClick is memoized, preventing ChildComponent re-render if it's the only prop changing
import React, { useCallback } from 'react';

const ParentComponentOptimized = ({ data }) => {
  const handleClick = useCallback(() => {
    // This function reference will only change if its dependencies change
    console.log('Button clicked');
  }, []); // Empty dependency array means it only gets created once

  return <ChildComponent onClick={handleClick} data={data} />;
};

Here's a simplified Next.js data fetching example that she might use to illustrate the difference:

// Example of Static Site Generation (SSG) with getStaticProps
// Ideal for blog posts, product pages where data doesn't change frequently
export async function getStaticProps() {
  const res = await fetch('https://api.example.com/posts');
  const posts = await res.json();
  return {
    props: {
      posts,
    },
    revalidate: 60, // Re-generate page every 60 seconds
  };
}

// Example of Server-Side Rendering (SSR) with getServerSideProps
// Ideal for dashboards, user-specific content where data changes frequently
export async function getServerSideProps(context) {
  const { req } = context;
  const res = await fetch(`https://api.example.com/user/${req.user.id}`);
  const userData = await res.json();
  return {
    props: {
      userData,
    },
  };
}
