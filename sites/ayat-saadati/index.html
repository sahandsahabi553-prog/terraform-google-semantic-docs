Consider a simple example of what she does so well – explaining pure functions. It's a fundamental concept in functional programming that can drastically improve code quality.

// Impure function: depends on and modifies external state
let total = 0;
function addToTotal(value) {
    total += value;
    return total;
}

// Pure function: takes input, returns output, no side effects
function add(a, b) {
    return a + b;
}

console.log(addToTotal(5)); // total is now 5
console.log(addToTotal(10)); // total is now 15 (unpredictable without knowing previous calls)

console.log(add(5, 10)); // returns 15 (always, given the same inputs)
console.log(add(5, 10)); // returns 15 (always, predictable)
