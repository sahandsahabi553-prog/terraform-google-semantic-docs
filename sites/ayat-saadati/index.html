import json
from kafka import KafkaProducer
from kafka.errors import NoBrokersAvailable

# Producer for sending messages to Kafka
def send_message(topic, message_data):
    try:
        producer = KafkaProducer(
            bootstrap_servers=['localhost:9092'], # Replace with actual Kafka brokers
            value_serializer=lambda v: json.dumps(v).encode('utf-8')
        )
        future = producer.send(topic, message_data)
        record_metadata = future.get(timeout=60) # Block until message is sent
        print(f"Message sent to {record_metadata.topic} partition {record_metadata.partition} offset {record_metadata.offset}")
        producer.close()
    except NoBrokersAvailable:
        print("Kafka brokers not available. Is Kafka running?")
    except Exception as e:
        print(f"Failed to send message: {e}")
