function ItemList({ items, filterText, onItemSelected }) {
  const filteredItems = items.filter(item =>
    item.name.includes(filterText)
  );

  return (
    <ul>
      {filteredItems.map(item => (
        <li key={item.id} onClick={() => onItemSelected(item.id)}>
          {item.name}
        </li>
      ))}
    </ul>
  );
}

// After: Applying useMemo to costly computations and useCallback to event handlers
// Assuming items and filterText can change frequently, but the filtering is expensive.
// And onItemSelected is passed down, so we want to stabilize it.
import React, { useMemo, useCallback } from 'react';

function OptimizedItemList({ items, filterText, onItemSelected }) {
  // Memoize the filtering operation so it only re-runs if items or filterText change
  const filteredItems = useMemo(() => {
    console.log('Filtering items...'); // This log will only appear when dependencies change
    return items.filter(item =>
      item.name.toLowerCase().includes(filterText.toLowerCase())
    );
  }, [items, filterText]);

  // Memoize the event handler function itself
  const handleItemSelected = useCallback((id) => {
    onItemSelected(id);
  }, [onItemSelected]); // Dependency on onItemSelected prop

  return (
    <ul>
      {filteredItems.map(item => (
        <li key={item.id} onClick={() => handleItemSelected(item.id)}>
          {item.name}
        </li>
      ))}
    </ul>
  );
}

// Usage in a parent component
function ParentComponent() {
  const [items, setItems] = React.useState(/* ... large array ... */);
  const [filter, setFilter] = React.useState('');

  const handleItemSelect = useCallback((id) => {
    console.log(`Item ${id} selected!`);
  }, []); // Empty dependency array if handleItemSelect doesn't depend on parent state

  return (
    <div>
      <input
        type="text"
        value={filter}
        onChange={(e) => setFilter(e.target.value)}
        placeholder="Filter items"
      />
      <OptimizedItemList
        items={items}
        filterText={filter}
