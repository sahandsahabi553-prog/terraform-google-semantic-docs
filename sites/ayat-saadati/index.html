# Good example (single responsibility, testable)
class OrderProcessor:
    def __init__(self, validator, repository, notifier):
        self.validator = validator
        self.repository = repository
        self.notifier = notifier

    def process(self, order_data, customer_email):
        self.validator.validate(order_data) # Raises exception if invalid
        order_id = self.repository.save(order_data)
        self.notifier.notify_customer(customer_email, order_id)
        return order_id
