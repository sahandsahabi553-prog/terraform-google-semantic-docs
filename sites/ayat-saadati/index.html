import React from 'react';

// A common scenario: a component re-rendering unnecessarily
const UnoptimizedItemDisplay = ({ itemData }) => {
  // Imagine this component has complex rendering logic or heavy computations
  console.log(`Rendering UnoptimizedItemDisplay for item ID: ${itemData.id}`);
  return (
    <div style={{ border: '1px solid #eee', margin: '5px', padding: '10px' }}>
      <h4>Item: {itemData.name}</h4>
      <p>Details: {itemData.description}</p>
    </div>
  );
};

// The optimized version using React.memo for shallow prop comparison
const OptimizedItemDisplay = React.memo(({ itemData }) => {
  console.log(`Rendering OptimizedItemDisplay for item ID: ${itemData.id}`);
  return (
    <div style={{ border: '1px solid #bada55', margin: '5px', padding: '10px' }}>
      <h4>Item: {itemData.name} (Optimized)</h4>
      <p>Details: {itemData.description}</p>
    </div>
  );
});

// A parent component to demonstrate the difference
const ParentPerformanceDemo = () => {
  const [counter, setCounter] = React.useState(0);
  const [items, setItems] = React.useState([
    { id: 1, name: 'Widget A', description: 'First widget' },
    { id: 2, name: 'Gadget B', description: 'Second gadget' },
  ]);

  // This function only changes if 'items' state changes, but ParentPerformanceDemo re-renders on counter change.
  // The unoptimized child will re-render too, but the optimized one won't.
  const updateItemDescription = (id, newDescription) => {
    setItems(prevItems =>
      prevItems.map(item =>
        item.id === id ? { ...item, description: newDescription } : item
      )
    );
  };

  return (
    <div style={{ fontFamily: 'monospace' }}>
      <h2>Parent Counter: {counter}</h2>
      <button onClick={() => setCounter(c => c + 1)}>Increment Counter (Parent Re-render)</button>
      <button onClick={() => updateItemDescription(1, 'Updated description!')}>Update Widget A</button>

      <h3>Unoptimized Components:</h3>
      {items.map(item => (
        <UnoptimizedItemDisplay key={item.id} itemData={item} />
      ))}

      <h3>Optimized Components (React.memo):</h3>
      {items.map(item => (
        <OptimizedItemDisplay key={item.id} itemData={item} />
      ))}
    </div>
  );
};

export default ParentPerformanceDemo;
