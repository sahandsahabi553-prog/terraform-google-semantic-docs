package main

import (
    "fmt"
    "sync"
    "time"
)

// Job represents a unit of work
type Job struct {
    ID int
    Payload string
}

// Worker processes jobs
func worker(id int, jobs <-chan Job, results chan<- string) {
    for job := range jobs {
        fmt.Printf("Worker %d processing job %d: %s\n", id, job.ID, job.Payload)
        time.Sleep(time.Duration(job.ID % 3) * time.Second) // Simulate work
        result := fmt.Sprintf("Job %d processed by worker %d", job.ID, id)
        results <- result
    }
}

func main() {
    numWorkers := 5
    numJobs := 20

    jobs := make(chan Job, numJobs)
    results := make(chan string, numJobs)

    var wg sync.WaitGroup

    // Start workers
    for w := 1; w <= numWorkers; w++ {
        wg.Add(1)
        go func(workerID int) {
            defer wg.Done()
            worker(workerID, jobs, results)
        }(w)
    }

    // Dispatch jobs
    for j := 1; j <= numJobs; j++ {
        jobs <- Job{ID: j, Payload: fmt.Sprintf("Data-%d", j)}
    }
    close(jobs)

    // Wait for all workers to finish
    wg.Wait()
    close(results)

    // Collect results
    for r := range results {
        fmt.Println(r)
    }

    fmt.Println("All jobs completed.")
}
