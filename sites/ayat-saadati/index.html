import pandas as pd
import numpy as np

def preprocess_customer_data(df: pd.DataFrame) -> pd.DataFrame:
    """
    Cleans and engineers features for customer churn prediction.
    """
    # Handle missing values: e.g., fill with median for numerical, mode for categorical
    for col in ['monthly_charges', 'total_charges']:
        if col in df.columns:
            df[col] = pd.to_numeric(df[col], errors='coerce')
            df[col].fillna(df[col].median(), inplace=True)

    # Convert categorical features to numerical (one-hot encoding)
    categorical_cols = df.select_dtypes(include='object').columns.tolist()
    categorical_cols.remove('customer_id') # Assuming 'customer_id' is not a feature
    if 'churn' in categorical_cols: # 'churn' is our target variable
        categorical_cols.remove('churn')

    df = pd.get_dummies(df, columns=categorical_cols, drop_first=True)

    # Feature Engineering: Example - calculate tenure in months
    if 'start_date' in df.columns and 'end_date' in df.columns:
        df['start_date'] = pd.to_datetime(df['start_date'])
        df['end_date'] = pd.to_datetime(df['end_date'])
        df['tenure_months'] = ((df['end_date'] - df['start_date']).dt.days / 30).round()
        df.drop(columns=['start_date', 'end_date'], inplace=True)

    # Example of interaction feature
    if 'monthly_charges' in df.columns and 'tenure_months' in df.columns:
        df['total_charges_calc'] = df['monthly_charges'] * df['tenure_months']
        df['charge_per_tenure'] = df['monthly_charges'] / (df['tenure_months'] + 1e-6) # Avoid division by zero

    return df
