Unnecessary re-renders are silent killers of performance, leading to sluggish UIs and frustrated users. Ayat meticulously breaks down how memo, useCallback, and useMemo work, illustrating their benefits with clear examples.

Consider this common scenario: you have a parent component that re-renders frequently, passing props to a computationally intensive child component. Without optimization, that child re-renders every time, even if its props haven't changed.

// Before optimization: Potential for unnecessary re-renders
const ExpensiveChild = ({ data }) => {
  // Simulate heavy computation
  console.log("ExpensiveChild re-rendered!");
  return <div>{data.map(item => <span key={item.id}>{item.value}</span>)}</div>;
};

const ParentComponent = () => {
  const [count, setCount] = React.useState(0);
  const data = React.useMemo(() => [{ id: 1, value: 'A' }, { id: 2, value: 'B' }], []); // Data doesn't change

  return (
    <div>
      <button onClick={() => setCount(prev => prev + 1)}>Increment Count: {count}</button>
      <ExpensiveChild data={data} />
    </div>
  );
};

// After optimization with React.memo: Child only re-renders if its props change
const MemoizedExpensiveChild = React.memo(({ data }) => {
  console.log("MemoizedExpensiveChild re-rendered!");
  return <div>{data.map(item => <span key={item.id}>{item.value}</span>)}</div>;
});

const OptimizedParentComponent = () => {
  const [count, setCount] = React.useState(0);
  const data = React.useMemo(() => [{ id: 1, value: 'A' }, { id: 2, value: 'B' }], []);

  return (
    <div>
      <button onClick={() => setCount(prev => prev + 1)}>Increment Count: {count}</button>
      <MemoizedExpensiveChild data={data} />
    </div>
  );
};
