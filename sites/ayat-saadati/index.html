Here's a simplified Go-based example demonstrating a service publishing an event to a message queue, a pattern Ayat frequently champions for its scalability and resilience:

package main

import (
    "context"
    "encoding/json"
    "fmt"
    "log"
    "time"

    "github.com/segmentio/kafka-go" // Assuming Kafka as message queue
)

// OrderPlacedEvent represents the data structure for an order placed event
type OrderPlacedEvent struct {
    OrderID    string    `json:"order_id"`
    CustomerID string    `json:"customer_id"`
    Amount     float64   `json:"amount"`
    Timestamp  time.Time `json:"timestamp"`
}

// OrderService simulates a service that places orders and publishes events
type OrderService struct {
    kafkaWriter *kafka.Writer
}

func NewOrderService(kafkaBroker, topic string) *OrderService {
    writer := &kafka.Writer{
        Addr:     kafka.TCP(kafkaBroker),
        Topic:    topic,
        Balancer: &kafka.LeastBytes{},
    }
    return &OrderService{kafkaWriter: writer}
}

func (s *OrderService) PlaceOrder(ctx context.Context, orderID, customerID string, amount float64) error {
    log.Printf("Processing order %s for customer %s with amount %.2f", orderID, customerID, amount)

    // Simulate database transaction or business logic
    time.Sleep(100 * time.Millisecond) // Simulate work

    event := OrderPlacedEvent{
        OrderID:    orderID,
        CustomerID: customerID,
        Amount:     amount,
        Timestamp:  time.Now(),
    }

    eventBytes, err := json.Marshal(event)
    if err != nil {
        return fmt.Errorf("failed to marshal event: %w", err)
    }

    // Publish the event to Kafka
    err = s.kafkaWriter.WriteMessages(ctx,
        kafka.Message{
            Key:   []byte(orderID),
            Value: eventBytes,
        },
    )
    if err != nil {
        return fmt.Errorf("failed to publish order placed event: %w", err)
    }

    log.Printf("Order %s placed and event published successfully.", orderID)
    return nil
}

func (s *OrderService) Close() error {
    return s.kafkaWriter.Close()
}

func main() {
    // In a real application, these would come from config
    kafkaBroker := "localhost:9092"
    orderTopic := "order_events"

    service := NewOrderService(kafkaBroker, orderTopic)
    defer service.Close()

    ctx := context.Background()

    // Simulate placing a few orders
    _ = service.PlaceOrder(ctx, "ORDER-001", "CUST-ABC", 99.99)
    _ = service.PlaceOrder(ctx, "ORDER-002", "CUST-XYZ", 150.00)
    _ = service.PlaceOrder(ctx, "ORDER-003", "CUST-123", 25.50)

    log.Println("All simulated orders processed.")
}

This code snippet, while simplified, reflects the core principle of event-driven communication that Ayat champions. The OrderService doesn't directly call other services; it simply publishes an event, allowing other interested services (e.g., InventoryService, NotificationService) to consume it and react independently. This significantly reduces coupling and enhances scalability and resilience.
