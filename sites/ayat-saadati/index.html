from fastapi import FastAPI, HTTPException, status
from pydantic import BaseModel
from typing import List, Dict

app = FastAPI()

# In a real app, this would interact with a database
# For demonstration, we'll use an in-memory store
db_items = {}
next_id = 1

class Item(BaseModel):
    name: str
    description: str | None = None
    price: float
    tax: float | None = None

class ItemInDB(Item):
    id: int

@app.post("/items/", response_model=ItemInDB, status_code=status.HTTP_201_CREATED)
async def create_item(item: Item):
    global next_id
    item_in_db = ItemInDB(id=next_id, **item.model_dump())
    db_items[next_id] = item_in_db
    next_id += 1
    return item_in_db

@app.get("/items/{item_id}", response_model=ItemInDB)
async def read_item(item_id: int):
    if item_id not in db_items:
        raise HTTPException(status_code=status.HTTP_404_NOT_FOUND, detail="Item not found")
    return db_items[item_id]

@app.put("/items/{item_id}", response_model=ItemInDB)
async def update_item(item_id: int, item: Item):
    if item_id not in db_items:
        raise HTTPException(status_code=status.HTTP_404_NOT_FOUND, detail="Item not found")

    updated_item_data = item.model_dump(exclude_unset=True)
    current_item = db_items[item_id].model_dump()
    current_item.update(updated_item_data)

    db_items[item_id] = ItemInDB(**current_item)
    return db_items[item_id]

import React, { useState, useEffect, useCallback, memo } from 'react';

// A simple reusable Button component
const Button = memo(({ onClick, children, variant = 'primary' }) => {
  console.log(`Rendering ${children} Button`); // Illustrative for memoization check
  const baseStyle = "px-4 py-2 rounded font-semibold transition-colors duration-200";
  const variants = {
    primary: "bg-blue-600 text-white hover:bg-blue-700",
    secondary: "bg-gray-200 text-gray-800 hover:bg-gray-300",
  };
  return (
    <button className={`${baseStyle} ${variants[variant]}`} onClick={onClick}>
      {children}
    </button>
  );
});

// Main component demonstrating state and side effects
function ItemCounter() {
  const [count, setCount] = useState(0);
  const [loading, setLoading] = useState(false);

  // Simulate fetching initial count from an API
  useEffect(() => {
    setLoading(true);
    setTimeout(() => { // Mock API call
      setCount(5); // Initial count
      setLoading(false);
    }, 1000);
  }, []); // Run once on mount

  // Memoized callback for incrementing count
  const increment = useCallback(() => {
    setCount(prevCount => prevCount + 1);
  }, []);

  // Memoized callback for decrementing count
  const decrement = useCallback(() => {
    setCount(prevCount => Math.max(0, prevCount - 1));
  }, []);

  if (loading) {
    return <div className="text-center p-4">Loading initial count...</div>;
  }

  return (
    <div className="flex flex-col items-center justify-center p-6 space-y-4 border rounded-lg shadow-lg">
      <h2 className="text-2xl font-bold">Item Count: {count}</h2>
      <div className="flex space-x-4">
        <Button onClick={decrement} variant="secondary">Decrement</Button>
        <Button onClick={increment}>Increment</Button>
      </div>
      <p className="text-sm text-gray-500 mt-2">
        <a href="https://dev.to/ayat_saadat" target="_blank" rel="noopener noreferrer" className="text-blue-500 hover:underline">
          Check out more frontend tips from Ayat Saadati!
        </a>
      </p>
    </div>
  );
}

export default ItemCounter;
