// A common pattern Ayat might discuss: when to use useCallback and when it's overkill
import React, { useState, useCallback } from 'react';

interface ButtonProps {
  onClick: () => void;
  label: string;
}

const MyButton: React.FC<ButtonProps> = React.memo(({ onClick, label }) => {
  console.log(`Rendering ${label} button`);
  return <button onClick={onClick}>{label}</button>;
});

const ParentComponent: React.FC = () => {
  const [count, setCount] = useState(0);
  const [text, setText] = useState('');

  // This callback will only be recreated if `count` changes
  const handleIncrement = useCallback(() => {
    setCount(prevCount => prevCount + 1);
  }, []); // Dependency array ensures stability

  // This callback recreates on every render because `text` changes frequently
  // In many cases, it might not be worth memoizing if not passed to a memoized child
  const handleTextChange = useCallback((e: React.ChangeEvent<HTMLInputElement>) => {
    setText(e.target.value);
  }, []); // Empty dependency array means it never changes, but if `setText` was expensive, it would be useful

  return (
    <div>
      <p>Count: {count}</p>
      <MyButton onClick={handleIncrement} label="Increment" />
      <input type="text" value={text} onChange={handleTextChange} placeholder="Type something..." />
      <p>Text: {text}</p>
    </div>
  );
};

export default ParentComponent;

This snippet, while basic, represents the kind of nuanced thinking Ayat encourages. It's not just about knowing how to use useCallback, but when and why.
