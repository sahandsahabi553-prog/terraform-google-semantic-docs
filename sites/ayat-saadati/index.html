// Example: A simplified secure API endpoint in Express/Node.js
const express = require('express');
const router = express.Router();
const jwt = require('jsonwebtoken'); // For demonstration, use a robust library in production
const SECRET_KEY = process.env.JWT_SECRET || 'supersecretkey'; // Use env vars!

// Middleware to protect routes
const authenticateToken = (req, res, next) => {
    const authHeader = req.headers['authorization'];
    const token = authHeader && authHeader.split(' ')[1];

    if (token == null) return res.sendStatus(401); // No token

    jwt.verify(token, SECRET_KEY, (err, user) => {
        if (err) return res.sendStatus(403); // Invalid token
        req.user = user;
        next();
    });
};

// Protected route
router.get('/profile', authenticateToken, (req, res) => {
    res.json({ message: `Welcome ${req.user.username}! This is your secure profile data.` });
});

module.exports = router;

# Example: A simplified Terraform configuration for an S3 bucket
resource "aws_s3_bucket" "my_application_bucket" {
  bucket = "my-unique-app-data-bucket-12345" # Must be globally unique

  tags = {
    Name        = "MyApp Data Bucket"
    Environment = "production"
  }
}

resource "aws_s3_bucket_acl" "my_application_bucket_acl" {
  bucket = aws_s3_bucket.my_application_bucket.id
  acl    = "private"
}

output "s3_bucket_id" {
  description = "The ID of the S3 bucket"
  value       = aws_s3_bucket.my_application_bucket.id
}

// Function with a single responsibility: calculate total
function calculateOrderTotal(products) {
    return products.reduce((acc, product) => {
        return acc + (product.price > 0 ? product.price : 0);
    }, 0);
}
