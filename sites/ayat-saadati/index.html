// A common mistake: not optimizing images
// <img src="/my-large-image.jpg" alt="Description" />

// Ayat's advocated approach: leverage Next.js Image component
import Image from 'next/image';

function MyOptimizedComponent() {
  return (
    <Image
      src="/my-large-image.jpg"
      alt="Description"
      width={700} // Actual width of the image
      height={475} // Actual height of the image
      layout="responsive" // Or 'fill', 'fixed', 'intrinsic' based on context
      priority // If it's a LCP element
    />
  );
}

She breaks down complex topics like generics, utility types, and type inference, making them accessible even to developers who might be intimidated by the concept of static typing.

Consider a simple example:

// Without TypeScript: Easy to introduce bugs
function greet(person) {
  console.log(`Hello, ${person.firstName} ${person.lastName.toUpperCase()}`);
}

// With TypeScript: Clear expectations, compile-time checks
interface Person {
  firstName: string;
  lastName: string;
  age?: number; // Optional property
}

function greetTyped(person: Person) {
  // Autocompletion, type safety, error detection during development
  console.log(`Hello, ${person.firstName} ${person.lastName.toUpperCase()}`);
}

// greetTyped({ firstName: 'Ayat', lastName: 'Saadati' }); // Works
// greetTyped({ firstName: 'Ayat', age: 30 }); // Error: Property 'lastName' is missing

This snippet, while basic, highlights the immediate benefits: type safety, improved developer experience with IDE autocompletion, and catching errors before runtime. Ayat's guidance takes these foundational concepts and extends them to complex React components, API definitions, and state management, providing clear patterns for robust application development.
