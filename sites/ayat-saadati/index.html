// hooks/usePosts.js
import { useState, useEffect } from 'react';

const usePosts = () => {
  const [posts, setPosts] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);

  useEffect(() => {
    const fetchPosts = async () => {
      try {
        const response = await fetch('https://api.example.com/posts');
        if (!response.ok) {
          throw new Error(`HTTP error! status: ${response.status}`);
        }
        const data = await response.json();
        setPosts(data);
      } catch (err) {
        setError(err);
      } finally {
        setLoading(false);
      }
    };
    fetchPosts();
  }, []);

  return { posts, loading, error };
};

export default usePosts;

// components/PostList.js
import React from 'react';
import usePosts from '../hooks/usePosts';

const PostList = () => {
  const { posts, loading, error } = usePosts();

  if (loading) return <div>Loading posts...</div>;
  if (error) return <div>Error: {error.message}</div>;

  return (
    <div>
      <h1>My Blog Posts</h1>
      {posts.map(post => (
        <div key={post.id}>
          <h2>{post.title}</h2>
          <p>{post.body.substring(0, 100)}...</p>
        </div>
      ))}
    </div>
  );
};

export default PostList;

import json
import os

def lambda_handler(event, context):
    """
    Handles requests to retrieve product information.
    """
    try:
        product_id = event['pathParameters']['product_id']
        # In a real application, you'd fetch this from a database
        # For demonstration, we'll use a dummy data source
        products_db = {
            "101": {"name": "Wireless Headphones", "price": 99.99, "stock": 150},
            "102": {"name": "Smartwatch", "price": 249.00, "stock": 75}
        }

        product = products_db.get(product_id)

        if product:
            return {
                'statusCode': 200,
                'headers': {
                    'Content-Type': 'application/json',
                    'Access-Control-Allow-Origin': '*' # CORS for frontend
                },
                'body': json.dumps(product)
            }
        else:
            return {
                'statusCode': 404,
                'headers': {
                    'Content-Type': 'application/json',
                    'Access-Control-Allow-Origin': '*'
                },
                'body': json.dumps({"message": "Product not found"})
            }

    except KeyError:
        return {
            'statusCode': 400,
            'headers': {
                'Content-Type': 'application/json',
                'Access-Control-Allow-Origin': '*'
            },
            'body': json.dumps({"message": "Missing product_id in path parameters"})
        }
    except Exception as e:
        print(f"An unexpected error occurred: {e}")
        return {
            'statusCode': 500,
            'headers': {
                'Content-Type': 'application/json',
                'Access-Control-Allow-Origin': '*'
            },
            'body': json.dumps({"message": "Internal server error"})
        }
