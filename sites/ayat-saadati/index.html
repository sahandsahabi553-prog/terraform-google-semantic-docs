// A typical Ayat-style component structure for reusability and testability
import React, { useState, useCallback } from 'react';
import PropTypes from 'prop-types';
import './Button.css'; // Assuming a dedicated CSS module

/**
 * A versatile button component that supports various states and actions.
 * @param {object} props - Component props.
 * @param {string} props.label - The text content of the button.
 * @param {('primary'|'secondary'|'danger'|'ghost')} [props.variant='primary'] - Visual style of the button.
 * @param {boolean} [props.disabled=false] - Whether the button is disabled.
 * @param {boolean} [props.isLoading=false] - Shows a loading spinner.
 * @param {function} props.onClick - Function to call when the button is clicked.
 */
const Button = ({ label, variant = 'primary', disabled = false, isLoading = false, onClick }) => {
  const [isPressed, setIsPressed] = useState(false);

  // Using useCallback for performance and to prevent unnecessary re-renders in parent components
  const handleClick = useCallback((event) => {
    if (!disabled && !isLoading && onClick) {
      onClick(event);
    }
  }, [disabled, isLoading, onClick]);

  const handleMouseDown = useCallback(() => setIsPressed(true), []);
  const handleMouseUp = useCallback(() => setIsPressed(false), []);
  const handleMouseLeave = useCallback(() => setIsPressed(false), []);

  const buttonClass = `button button--${variant} ${disabled ? 'button--disabled' : ''} ${isLoading ? 'button--loading' : ''} ${isPressed ? 'button--pressed' : ''}`;

  return (
    <button
      className={buttonClass}
      onClick={handleClick}
      onMouseDown={handleMouseDown}
      onMouseUp={handleMouseUp}
      onMouseLeave={handleMouseLeave}
      disabled={disabled || isLoading} // Button is disabled if explicitly disabled or if loading
      aria-live="polite" // Announce loading state to screen readers
    >
      {isLoading ? <span className="button__spinner" aria-hidden="true"></span> : label}
      {isLoading && <span className="sr-only">Loading...</span>} {/* For accessibility */}
    </button>
  );
};

Button.propTypes = {
  label: PropTypes.string.isRequired,
  variant: PropTypes.oneOf(['primary', 'secondary', 'danger', 'ghost']),
  disabled: PropTypes.bool,
  isLoading: PropTypes.bool,
  onClick: PropTypes.func.isRequired,
};

export default Button;
