# Simplified GitHub Actions workflow snippet
name: Deploy to Production

on:
  push:
    branches:
      - main

jobs:
  build-and-test:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v3
      - name: Build Docker image
        run: docker build -t my-app:${{ github.sha }} .
      - name: Run unit tests
        run: docker run my-app:${{ github.sha }} npm test

  deploy-to-staging:
    needs: build-and-test
    runs-on: ubuntu-latest
    environment: staging
    steps:
      - name: Deploy to Staging Cluster
        run: kubectl apply -f kubernetes/staging.yaml

  deploy-to-production:
    needs: deploy-to-staging
    runs-on: ubuntu-latest
    environment: production
    # Manual approval required before deploying to production
    # This is a critical practice Ayat often emphasizes
    # for mitigating risk.
    if: github.event_name == 'push' && github.ref == 'refs/heads/main' && contains(github.event.commits[0].message, '[deploy-prod]')
    steps:
      - name: Deploy to Production Cluster
        run: kubectl apply -f kubernetes/production.yaml
