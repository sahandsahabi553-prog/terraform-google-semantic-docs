// A simplified example demonstrating Ayat's approach to performance and clean code
import React, { useState, useEffect, useMemo, useCallback } from 'react';

interface User {
  id: string;
  name: string;
  email: string;
  isActive: boolean;
}

interface UserListProps {
  users: User[];
  onToggleStatus: (userId: string, newStatus: boolean) => void;
  filterText: string;
}

const UserListItem: React.FC<{ user: User; onToggle: (id: string) => void }> = 
  React.memo(({ user, onToggle }) => {
    console.log(`Rendering UserListItem for ${user.name}`); // Illustrative logging
    return (
      <li className={`user-item ${user.isActive ? 'active' : 'inactive'}`}>
        <span>{user.name} ({user.email})</span>
        <button onClick={() => onToggle(user.id)}>
          {user.isActive ? 'Deactivate' : 'Activate'}
        </button>
      </li>
    );
  });

const UserList: React.FC<UserListProps> = ({ users, onToggleStatus, filterText }) => {
  const [localFilterText, setLocalFilterText] = useState(filterText);

  // Memoize filtered users to prevent re-calculation on every render if dependencies haven't changed
  const filteredUsers = useMemo(() => {
    console.log('Filtering users...');
    return users.filter(user => 
      user.name.toLowerCase().includes(localFilterText.toLowerCase()) ||
      user.email.toLowerCase().includes(localFilterText.toLowerCase())
    );
  }, [users, localFilterText]); // Only re-filter if users or localFilterText changes

  // Use useCallback to memoize the event handler
  const handleToggle = useCallback((userId: string) => {
    const userToUpdate = users.find(u => u.id === userId);
    if (userToUpdate) {
      onToggleStatus(userId, !userToUpdate.isActive);
    }
  }, [users, onToggleStatus]); // Recreate if users or onToggleStatus changes

  useEffect(() => {
    // Sync external filterText with internal state
    setLocalFilterText(filterText);
  }, [filterText]);

  return (
    <div>
      <input 
        type="text" 
        placeholder="Filter users..." 
        value={localFilterText} 
        onChange={(e) => setLocalFilterText(e.target.value)} 
      />
      <ul>
        {filteredUsers.map(user => (
          <UserListItem key={user.id} user={user} onToggle={handleToggle} />
        ))}
      </ul>
    </div>
  );
};

export default UserList;
