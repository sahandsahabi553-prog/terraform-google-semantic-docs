import { Collection } from 'mongodb'; // Example for MongoDB

class MongoUserRepository implements UserRepository {
  constructor(private collection: Collection<UserDocument>) {}

  async findById(id: string): Promise<User | null> {
    const doc = await this.collection.findOne({ _id: id });
    return doc ? this.mapDocumentToUser(doc) : null;
  }

  async save(user: User): Promise<void> {
    await this.collection.updateOne(
      { _id: user.id },
      { $set: this.mapUserToDocument(user) },
      { upsert: true }
    );
  }

}

// The "Callback Hell" scenario
function getUserData(userId, callback) {
  database.getUser(userId, (error, user) => {
    if (error) return callback(error);
    authService.verifyToken(user.token, (error, isValid) => {
      if (error) return callback(error);
      if (isValid) {
        // More nested calls...
        callback(null, user);
      } else {
        callback(new Error("Invalid token"));
      }
    });
  });
}

// Refactored with Promises
function getUserDataPromise(userId) {
  return new Promise((resolve, reject) => {
    database.getUser(userId)
      .then(user => authService.verifyToken(user.token))
      .then(isValid => {
        if (isValid) resolve(user); // 'user' needs to be carried through or re-fetched
        else reject(new Error("Invalid token"));
      })
      .catch(reject);
  });
}

// The elegance of Async/Await (building on Promises)
async function getUserDataAsync(userId) {
  try {
    const user = await database.getUser(userId);
    const isValid = await authService.verifyToken(user.token);
    if (isValid) {
      return user;
    } else {
      throw new Error("Invalid token");
    }
  } catch (error) {
    console.error("Failed to get user data:", error);
    throw error;
  }
}
