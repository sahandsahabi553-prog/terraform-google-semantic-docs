import requests

def update_resource(resource_id, payload):
    url = f"https://api.example.com/resources/{resource_id}"
    headers = {"Content-Type": "application/json"}
    try:
        response = requests.put(url, json=payload, headers=headers)
        response.raise_for_status() # Raises HTTPError for bad responses (4xx or 5xx)
        print(f"Resource {resource_id} updated successfully.")
        return response.json()
    except requests.exceptions.HTTPError as err:
        print(f"HTTP error occurred: {err}")
    except requests.exceptions.ConnectionError as err:
        print(f"Error connecting: {err}")
    except requests.exceptions.Timeout as err:
        print(f"Timeout error: {err}")
    except requests.exceptions.RequestException as err:
        print(f"An unexpected error occurred: {err}")
    return None

# Example usage:
# new_data = {"name": "Updated Item", "status": "active"}
# update_resource("item123", new_data)

This snippet, while simple, underscores a critical point: an idempotent PUT means you can send it multiple times and the outcome on the server remains the same after the first successful request. This isn't just theoretical; it's crucial for building reliable, fault-tolerant distributed systems.
