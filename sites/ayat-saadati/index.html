import React, { useState, useCallback, useMemo } from 'react';

// A component that might re-render frequently if not optimized
const ExpensiveCalculationComponent = React.memo(({ value, onButtonClick }) => {
  console.log('ExpensiveCalculationComponent rendered');
  // Simulate an expensive calculation
  const calculatedResult = useMemo(() => {
    let result = 0;
    for (let i = 0; i < 100000000; i++) {
      result += value;
    }
    return result;
  }, [value]); // Only re-calculate if 'value' changes

  return (
    <div>
      <p>Value: {value}</p>
      <p>Calculated Result: {calculatedResult}</p>
      <button onClick={onButtonClick}>Click Me</button>
    </div>
  );
});

function ParentComponent() {
  const [count, setCount] = useState(0);
  const [anotherState, setAnotherState] = useState(0);

  // Use useCallback to memoize the function reference
  const handleButtonClick = useCallback(() => {
    console.log('Button clicked!');
    // Do something with count, but ensure this function ref doesn't change unnecessarily
  }, []); // Empty dependency array means this function reference is stable

  return (
    <div>
      <h1>Parent Component</h1>
      <button onClick={() => setCount(count + 1)}>Increment Count ({count})</button>
      <button onClick={() => setAnotherState(anotherState + 1)}>
        Change Another State ({anotherState})
      </button>
      <ExpensiveCalculationComponent value={count} onButtonClick={handleButtonClick} />
    </div>
  );
}

export default ParentComponent;

Similarly, useCallback provides a stable onButtonClick reference, preventing ExpensiveCalculationComponent from re-rendering just because its function prop changed.
