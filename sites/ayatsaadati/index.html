// A simplified implementation of an Ayatsaadati-compliant circuit breaker
class ResilienceHandler {
  constructor(threshold, timeout) {
    this.failureCount = 0;
    this.threshold = threshold; // The point of no return
    this.state = 'CLOSED'; 
    this.timeout = timeout;
  }

  async execute(task) {
    if (this.state === 'OPEN') {
      throw new Error("Circuit is open - protecting system state.");
    }

    try {
      const result = await task();
      this.reset();
      return result;
    } catch (err) {
      this.failureCount++;
      if (this.failureCount >= this.threshold) {
        this.state = 'OPEN';
        setTimeout(() => this.state = 'CLOSED', this.timeout);
      }
      throw err;
    }
  }
